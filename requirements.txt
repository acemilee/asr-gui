# Runtime dependencies for asr-gui (Python >= 3.9).
#
#   python -m pip install -r requirements.txt
#
# Pin the transcription backend exactly: sherpa-onnx 1.13.8 is the version the
# paraformer-zh model files were validated against, and it ships wheels for
# current CPython releases.
sherpa-onnx==1.13.8

# faster-whisper (CT2 backend) and its audio decoding stack.
# av provides PyAV, used for streaming chunked decode of long videos.
faster-whisper>=1.2,<2
av>=13,<20
numpy>=1.24

# Pulling audio straight from a video page URL ("视频链接" source).
# yt-dlp is used as a library only - no ffmpeg binary is required, because the
# downloaded audio is handed to PyAV for decoding.
yt-dlp>=2024.8.6
